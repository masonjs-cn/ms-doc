基本用法 ​

<template>
  <div>
    <el-button @click="open">打开</el-button>

    <MsDialog v-model:visible="visible" title="标题" width="500px">
      <span>这是一段信息</span>
    </MsDialog>
  </div>
</template>

<script setup lang="ts">
import { ref } from "vue";
const visible = ref();

const open = () => {
  visible.value = true;
};
</script>
